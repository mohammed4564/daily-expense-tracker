// ====================================== // FORMAT CURRENCY FOR PDF // ====================================== function pdfMoney(amount) { const n = Number(amount); if (!Number.isFinite(n)) return "INR 0.00"; return "INR " + n.toLocaleString("en-IN", { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } // ====================================== // GET LOGGED-IN USER INFORMATION // ====================================== function getReportUserInfo() { try { const storedUser = localStorage.getItem("user"); if (!storedUser) return {}; const user = typeof storedUser === "string" ? JSON.parse(storedUser) : storedUser; return { fullName: user.full_name || user.name || "Not available", username: user.username || "Not available", email: user.email || "Not available", userId: user.user_id || user.id || "Not available" }; } catch (error) { console.error("Unable to read user information:", error); return {}; } } // ====================================== // DRAW PDF HEADER BAR // ====================================== function drawPdfHeader(doc, pageWidth) { // Dark blue banner doc.setFillColor(36, 52, 71); // #243447 doc.rect(0, 0, pageWidth, 32, "F"); // App name (left) doc.setTextColor(255, 255, 255); doc.setFont("helvetica", "bold"); doc.setFontSize(18); doc.text("Daily Expense Tracker", 14, 14); // Subtitle (left, below) doc.setFont("helvetica", "normal"); doc.setFontSize(10); doc.setTextColor(213, 220, 227); doc.text("Expense Report", 14, 22); // Generated date (right, aligned) doc.setFontSize(9); doc.setTextColor(213, 220, 227); doc.text( `Generated: ${dateString(new Date())}`, pageWidth - 14, 14, { align: "right" } ); // Reset text color for body doc.setTextColor(0, 0, 0); } // ====================================== // DRAW SECTION TITLE // ====================================== function drawSectionTitle(doc, title, y) { doc.setFont("helvetica", "bold"); doc.setFontSize(12); doc.setTextColor(36, 52, 71); doc.text(title, 14, y); doc.setTextColor(0, 0, 0); // Thin underline doc.setDrawColor(37, 99, 235); doc.setLineWidth(0.5); doc.line(14, y + 2, 60, y + 2); } // ====================================== // DOWNLOAD PDF REPORT // ====================================== function downloadPDFReport() { try { // ---- Library checks ---- if (!window.jspdf || !window.jspdf.jsPDF) { alert("jsPDF library failed to load. Please check your internet connection and refresh the page."); return; } const { jsPDF } = window.jspdf; const doc = new jsPDF({ unit: "mm", format: "a4" }); if (typeof doc.autoTable !== "function") { alert("PDF table plugin failed to load. Please refresh the page."); return; } // ---- Period check ---- if (!startDate || !endDate) { alert("Please select a reporting period first."); return; } const items = getExpensesInRange(startDate, endDate); if (!items.length) { alert("No expenses found for the selected period."); return; } // ---- Compute stats ---- const total = getTotal(items); const average = total / items.length; const highest = items.reduce((max, item) => Number(item.amount || 0) > Number(max?.amount || 0) ? item : max, null ); const previousTotal = getTotal(getPreviousRangeExpenses()); let comparison = "No previous-period spending"; if (previousTotal > 0) { const change = ((total - previousTotal) / previousTotal) * 100; comparison = `${Math.abs(change).toFixed(1)}% ` + (change > 0 ? "higher" : change < 0 ? "lower" : "unchanged"); } const userInfo = getReportUserInfo(); const pageWidth = doc.internal.pageSize.getWidth(); const pageHeight = doc.internal.pageSize.getHeight(); const margin = 14; // ====================================== // HEADER // ====================================== drawPdfHeader(doc, pageWidth); let y = 42; // ====================================== // REPORT + USER INFO (2 COLUMNS) // ====================================== drawSectionTitle(doc, "Report Information", y); y += 7; doc.setFont("helvetica", "normal"); doc.setFontSize(10); // Left column: report period doc.setFont("helvetica", "bold"); doc.text("Report Period:", margin, y); doc.setFont("helvetica", "normal"); doc.text( `${dateString(startDate)} to ${dateString(endDate)}`, margin + 32, y ); // Right column: user name doc.setFont("helvetica", "bold"); doc.text("User:", pageWidth / 2 + 5, y); doc.setFont("helvetica", "normal"); doc.text( String(userInfo.fullName || "Not available"), pageWidth / 2 + 22, y ); y += 7; // Row 2: transactions + username doc.setFont("helvetica", "bold"); doc.text("Transactions:", margin, y); doc.setFont("helvetica", "normal"); doc.text(String(items.length), margin + 32, y); doc.setFont("helvetica", "bold"); doc.text("Username:", pageWidth / 2 + 5, y); doc.setFont("helvetica", "normal"); doc.text( String(userInfo.username || "Not available"), pageWidth / 2 + 22, y ); y += 7; // Row 3: total spending + email doc.setFont("helvetica", "bold"); doc.text("Total Spending:", margin, y); doc.setFont("helvetica", "normal"); doc.text(pdfMoney(total), margin + 32, y); doc.setFont("helvetica", "bold"); doc.text("Email:", pageWidth / 2 + 5, y); doc.setFont("helvetica", "normal"); doc.text( String(userInfo.email || "Not available"), pageWidth / 2 + 22, y ); y += 12; // ====================================== // EXPENSE SUMMARY TABLE // ====================================== drawSectionTitle(doc, "Expense Summary", y); y += 5; doc.autoTable({ startY: y, head: [["Metric", "Value"]], body: [ ["Total Spending", pdfMoney(total)], ["Number of Transactions", String(items.length)], ["Average Expense", pdfMoney(average)], ["Highest Expense", pdfMoney(highest?.amount || 0)], ["Highest Expense Date", highest ? String(highest.expense_date).slice(0, 10) : "No records"], ["Previous Period Comparison", comparison] ], theme: "grid", headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: "bold", halign: "left" }, styles: { fontSize: 9, cellPadding: 3, valign: "middle" }, columnStyles: { 0: { cellWidth: 70, fontStyle: "bold" }, 1: { cellWidth: "auto", halign: "right" } }, margin: { left: margin, right: margin } }); y = doc.lastAutoTable.finalY + 12; // ====================================== // CATEGORY-WISE SPENDING // ====================================== const categoryTotals = {}; items.forEach(exp => { const cat = exp.category || "Other"; categoryTotals[cat] = (categoryTotals[cat] || 0) + Number(exp.amount || 0); }); const categoryRows = Object.entries(categoryTotals) .sort((a, b) => b[1] - a[1]) .map(([cat, amount]) => [ cat, pdfMoney(amount), total > 0 ? `${((amount / total) * 100).toFixed(1)}%` : "0.0%" ]); if (y > pageHeight - 40) { doc.addPage(); y = 20; } drawSectionTitle(doc, "Category-wise Spending", y); y += 5; doc.autoTable({ startY: y, head: [["Category", "Amount", "Share"]], body: categoryRows, theme: "striped", headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: "bold", halign: "left" }, styles: { fontSize: 9, cellPadding: 3 }, columnStyles: { 0: { cellWidth: 70 }, 1: { cellWidth: "auto", halign: "right" }, 2: { cellWidth: 30, halign: "right" } }, margin: { left: margin, right: margin } }); y = doc.lastAutoTable.finalY + 12; // ====================================== // TRANSACTION DETAILS // ====================================== const transactionRows = [...items] .sort((a, b) => String(a.expense_date).localeCompare(String(b.expense_date)) ) .map(exp => [ String(exp.expense_date || "").slice(0, 10), exp.category || "Other", exp.description || "-", pdfMoney(exp.amount) ]); if (y > pageHeight - 40) { doc.addPage(); y = 20; } drawSectionTitle(doc, "Expense Transaction Details", y); y += 5; doc.autoTable({ startY: y, head: [["Date", "Category", "Description", "Amount"]], body: transactionRows, theme: "striped", headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: "bold", halign: "left" }, styles: { fontSize: 8, cellPadding: 2.5, overflow: "linebreak", valign: "top" }, columnStyles: { 0: { cellWidth: 25 }, // Date 1: { cellWidth: 30 }, // Category 2: { cellWidth: "auto" }, // Description (flex) 3: { cellWidth: 30, halign: "right" } // Amount (right-aligned) }, margin: { left: margin, right: margin }, didDrawPage: function () { // Re-draw header on every new page drawPdfHeader(doc, pageWidth); } }); // ====================================== // FOOTER ON EVERY PAGE // ====================================== const pageCount = doc.internal.getNumberOfPages(); for (let page = 1; page <= pageCount; page++) { doc.setPage(page); // Line above footer doc.setDrawColor(220, 224, 230); doc.setLineWidth(0.3); doc.line(margin, pageHeight - 14, pageWidth - margin, pageHeight - 14); doc.setFont("helvetica", "normal"); doc.setFontSize(8); doc.setTextColor(119, 119, 119); // Left: app name doc.text("Daily Expense Tracker", margin, pageHeight - 8); // Center: period doc.text( `${dateString(startDate)} to ${dateString(endDate)}`, pageWidth / 2, pageHeight - 8, { align: "center" } ); // Right: page number doc.text( `Page ${page} of ${pageCount}`, pageWidth - margin, pageHeight - 8, { align: "right" } ); } // ====================================== // SAVE // ====================================== const fileName = `Expense_Report_${dateString(startDate)}_to_${dateString(endDate)}.pdf`; doc.save(fileName); console.log("PDF download started:", fileName); } catch (error) { console.error("PDF generation error:", error); alert("Unable to generate PDF. Open browser F12 Console and check the error."); } }
